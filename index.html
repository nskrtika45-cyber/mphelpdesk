rules_version = '2'; // ============================================================ // MP HelpDesk — Firestore Security Rules (HelpDesk V2: roles + departments) // ============================================================ // OPTIONAL and NOT TESTED against your live project. // The app works without changing your rules — the department-wise // visibility is already enforced by the queries the app runs. These // rules add a second line of defence so the same limits also hold if // someone calls Firebase directly. // // BEFORE pasting this over your current rules (Firebase Console → // Firestore Database → Rules), open your current rules and compare. // Then use the "Rules Playground" / publish to a quiet moment and // check that: login works, an Employee can raise and see own tickets, // a Ticket Owner can see the department's tickets. // If anything breaks, press "Restore" to the previous version in the // Rules history. // // Roles stored in users/{uid}.role: // User (Employee) · TicketOwner (Department SPOC) · Admin (Administrator) // // What the ticket rules enforce // READ Admin: all. Employee: own tickets. // Ticket Owner: own tickets + tickets whose assignedDeptId is // their department. // CREATE createdAt must be the real server time; the ticket must be // raised by the signed-in user (Admin may raise for others). // UPDATE only someone who can read the ticket, and createdAt can // never change after creation. // DELETE Admin only. // ============================================================ service cloud.firestore { match /databases/{database}/documents { function isSignedIn() { return request.auth != null; } function me() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; } function isAdmin() { return isSignedIn() && me().role == 'Admin'; } // Field must be unchanged from the stored document, when it exists there. function unchanged(field) { return !(field in resource.data) || request.resource.data[field] == resource.data[field]; } // May the signed-in user read the ticket in `t`? function canReadTicket(t) { return isSignedIn() && ( me().role == 'Admin' || t.userId == request.auth.uid || (me().role == 'TicketOwner' && 'assignedDeptId' in t && t.assignedDeptId == me().deptId) ); } match /tickets/{ticketId} { allow read: if canReadTicket(resource.data); allow create: if isSignedIn() && (request.resource.data.userId == request.auth.uid || isAdmin()) && request.resource.data.createdAt == request.time; allow update: if canReadTicket(resource.data) && unchanged('createdAt'); allow delete: if isAdmin(); } match /users/{userId} { allow read: if isSignedIn(); allow create: if isSignedIn(); // admin-created accounts / first-login profile // Admin can change anything; a user can edit their own record but never their role or department. allow update: if isAdmin() || (request.auth.uid == userId && request.resource.data.role == resource.data.role && request.resource.data.deptId == resource.data.deptId); allow delete: if isAdmin(); } // Masters — everyone signed in can read, only Admin can change match /issueTypes/{id} { allow read: if isSignedIn(); allow write: if isAdmin(); } match /subTypes/{id} { allow read: if isSignedIn(); allow write: if isAdmin(); } match /priorities/{id} { allow read: if isSignedIn(); allow write: if isAdmin(); } match /departments/{id} { allow read: if isSignedIn(); allow write: if isAdmin(); } match /statuses/{id} { allow read: if isSignedIn(); allow write: if isAdmin(); } match /categories/{id} { allow read: if isSignedIn(); allow write: if isAdmin(); } match /slaSettings/{id} { allow read: if isSignedIn(); allow write: if isAdmin(); } // Page Access + Configure Fields settings (config/pageAccess, config/ticketFields) match /config/{id} { allow read: if isSignedIn(); allow write: if isAdmin(); } // One active login per user: each person may read/write only their own session record match /sessions/{uid} { allow read, write: if isSignedIn() && request.auth.uid == uid; allow read, delete: if isAdmin(); } match /emailQueue/{id} { allow read: if isAdmin(); allow create: if isSignedIn(); allow update, delete: if isAdmin(); } match /counters/{id} { allow read: if isSignedIn(); allow write: if isSignedIn(); // needed for the atomic ticket-number transaction } } }
